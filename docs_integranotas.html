Exemplo de Chamada Python

import requests

def verificar_sefaz(uf: str) -> dict:
    """Verifica status da SEFAZ para um estado. Não requer auth."""
    resp = requests.get("https://api.integranotas.com.br/v1/monitor", timeout=10)
    resp.raise_for_status()
    data = resp.json()
    # Filtrar pelo estado desejado
    return next((s for s in data if s.get("uf") == uf), None)

# faturamento/services/integranotas.py — adicionar método:

def emitir_cte(self, ct) -> dict:
    """Emite CT-e na SEFAZ via IntegraNotas."""
    payload = {
        "serie": ct.serie,
        "numero": str(ct.numero),
        "data_emissao": ct.data_emissao.isoformat(),
        "remetente": self._build_remetente_cte(ct),
        "destinatario": self._build_destinatario_cte(ct),
        "tomador": self._build_tomador_cte(ct),
        "valores": {"valor_total_servico": float(ct.valor_servico)},
        "nfe_vinculadas": [{"chave": doc.chave_nfe} for doc in ct.documentos.all()],
        "modal_rodoviario": self._build_modal_rodoviario(ct),
    }
    resp = self._post("/cte", payload)
    if resp.get("codigo") == 5023:
        # Processando em lote — aguardar e consultar
        import time; time.sleep(3)
        return self._get(f"/cte/{resp['chave']}")
    return resp

# faturamento/views.py — expandir view existente:

@csrf_exempt
@require_POST
def integranotas_webhook(request):
    """Recebe callbacks assíncronos do IntegraNotas para NF-e, CT-e e MDF-e."""
    try:
        data = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({"erro": "JSON inválido"}, status=400)

    tipo = data.get("tipo")           # "nfe", "cte", "mdfe"
    chave = data.get("chave")
    status = data.get("status")       # "autorizado", "cancelado", "rejeitado"
    xml = data.get("xml")             # BASE64
    pdf = data.get("pdf")             # BASE64

    if tipo == "nfe":
        nf = NotaFiscal.objects.filter(chave_acesso=chave).first()
        if nf and status == "autorizado":
            nf.status = "emitida"
            nf.xml_autorizacao = xml
            nf.pdf_danfe = pdf
            nf.protocolo_autorizacao = data.get("protocolo")
            nf.save()

    elif tipo == "cte":   # 🔴 ADICIONAR
        ct = ConhecimentoTransporte.objects.filter(chave_acesso=chave).first()
        if ct and status == "autorizado":
            ct.status = "emitido"
            ct.xml_autorizacao = xml
            ct.pdf_dacte = pdf
            ct.protocolo = data.get("protocolo")
            ct.save()

    elif tipo == "mdfe":  # 🔴 ADICIONAR
        mdfe = ManifestoDocumentoFiscal.objects.filter(chave_acesso=chave).first()
        if mdfe and status == "autorizado":
            mdfe.status = "emitido"
            mdfe.xml_autorizacao = xml
            mdfe.pdf_damdfe = pdf
            mdfe.protocolo = data.get("protocolo")
            mdfe.save()

    return JsonResponse({"ok": True})
