============================================================================= # CRUD DE VENDEDOR # ============================================================================= @login_required def vendedor_list(request): """ Lista vendedores da empresa com filtros e estatísticas GET /cadastros/vendedores/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') # Formulário de filtros filtro_form = VendedorFiltroForm(request.GET or None) # QuerySet base vendedores = Vendedor.objects.filter(empresa=empresa).select_related('supervisor', 'empresa') # Aplicar filtros se o formulário for válido if filtro_form.is_valid(): busca = filtro_form.cleaned_data.get('busca') status = filtro_form.cleaned_data.get('status') tipo_mercado = filtro_form.cleaned_data.get('tipo_mercado') if busca: vendedores = vendedores.filter( Q(nome_completo__icontains=busca) | Q(cpf__icontains=busca) | Q(email_corporativo__icontains=busca) ) if status: vendedores = vendedores.filter(status=status) if tipo_mercado: vendedores = vendedores.filter(tipo_mercado=tipo_mercado) # Estatísticas para o dashboard total_vendedores = vendedores.count() total_ativos = vendedores.filter(status='ATIVO').count() total_mercado_interno = vendedores.filter(tipo_mercado__in=['INTERNO', 'AMBOS']).count() total_exportacao = vendedores.filter(tipo_mercado__in=['EXTERNO', 'AMBOS']).count() context = { 'vendedores': vendedores, 'filtro_form': filtro_form, 'total_vendedores': total_vendedores, 'total_ativos': total_ativos, 'total_mercado_interno': total_mercado_interno, 'total_exportacao': total_exportacao, } return render(request, 'cadastros/vendedor_list.html', context) @login_required def vendedor_create(request): """ Cria novo vendedor GET/POST /cadastros/vendedores/novo/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') if request.method == 'POST': form = VendedorForm( request.POST, request.FILES, usuario=request.user, # ✅ Parâmetro correto empresa=empresa ) if form.is_valid(): vendedor = form.save() messages.success(request, f"✅ Vendedor '{vendedor.nome_completo}' cadastrado com sucesso!") return redirect('cadastros:vendedor_list') else: messages.error(request, '❌ Corrija os erros no formulário.') else: form = VendedorForm( usuario=request.user, # ✅ Parâmetro correto empresa=empresa ) context = { 'form': form, 'action': 'Criar', 'title': 'Novo Vendedor' } return render(request, 'cadastros/vendedor_form.html', context) @login_required def vendedor_update(request, pk): """ Edita vendedor existente GET/POST /cadastros/vendedores//editar/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') vendedor = get_object_or_404(Vendedor, pk=pk, empresa=empresa) if request.method == 'POST': form = VendedorForm( request.POST, request.FILES, instance=vendedor, usuario=request.user, # ✅ Parâmetro correto empresa=empresa ) if form.is_valid(): vendedor = form.save() messages.success(request, f"✅ Vendedor '{vendedor.nome_completo}' atualizado com sucesso!") return redirect('cadastros:vendedor_list') else: messages.error(request, '❌ Corrija os erros no formulário.') else: form = VendedorForm( instance=vendedor, usuario=request.user, # ✅ Parâmetro correto empresa=empresa ) context = { 'form': form, 'vendedor': vendedor, 'action': 'Editar', 'title': f'Editar: {vendedor.nome_completo}' } return render(request, 'cadastros/vendedor_form.html', context) @login_required def vendedor_detail(request, pk): """ Exibe detalhes completos do vendedor GET /cadastros/vendedores// """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') vendedor = get_object_or_404( Vendedor.objects.select_related('supervisor', 'empresa'), pk=pk, empresa=empresa ) # Busca subordinados (se for gerente) subordinados = [] if hasattr(vendedor, 'subordinados'): subordinados = vendedor.subordinados.filter(status='ATIVO') # Calcula totais de comissões comissoes_pendentes = vendedor.comissoes.filter(status='PENDENTE').aggregate( total=Sum('valor_comissao') )['total'] or 0 comissoes_pagas = vendedor.comissoes.filter(status='PAGA').aggregate( total=Sum('valor_comissao') )['total'] or 0 # Últimas 5 comissões ultimas_comissoes = vendedor.comissoes.all().order_by('-referencia_mes')[:5] context = { 'vendedor': vendedor, 'subordinados': subordinados, 'comissoes_pendentes': comissoes_pendentes, 'comissoes_pagas': comissoes_pagas, 'ultimas_comissoes': ultimas_comissoes, } return render(request, 'cadastros/vendedor_detail.html', context) @login_required def vendedor_delete(request, pk): """ Exclui vendedor (com validações de segurança) GET/POST /cadastros/vendedores//excluir/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') vendedor = get_object_or_404(Vendedor, pk=pk, empresa=empresa) # ✅ VALIDAÇÃO 1: Verifica se tem subordinados if hasattr(vendedor, 'subordinados') and vendedor.subordinados.exists(): total_subordinados = vendedor.subordinados.count() messages.error( request, f"❌ Não é possível excluir! O vendedor '{vendedor.nome_completo}' possui " f"{total_subordinados} subordinado(s). Reatribua-os primeiro." ) return redirect('cadastros:vendedor_list') # ✅ VALIDAÇÃO 2: Verifica comissões pendentes comissoes_pendentes = vendedor.comissoes.filter(status='PENDENTE').count() if comissoes_pendentes > 0: messages.error( request, f"❌ Não é possível excluir! O vendedor possui {comissoes_pendentes} " f"comissão(ões) pendente(s). Resolva-as primeiro." ) return redirect('cadastros:vendedor_list') # Confirmação de exclusão if request.method == 'POST': nome = vendedor.nome_completo vendedor.delete() messages.success(request, f"🗑️ Vendedor '{nome}' excluído com sucesso!") return redirect('cadastros:vendedor_list') context = {'vendedor': vendedor} return render(request, 'cadastros/vendedor_confirm_delete.html', context) # ============================================================================= # AÇÕES RÁPIDAS # ============================================================================= @login_required def vendedor_inativar(request, pk): """ Inativa vendedor (sem excluir do banco) POST /cadastros/vendedores//inativar/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') vendedor = get_object_or_404(Vendedor, pk=pk, empresa=empresa) if vendedor.status == 'ATIVO': vendedor.status = 'INATIVO' vendedor.data_desligamento = timezone.now().date() vendedor.save() messages.success(request, f"✅ Vendedor '{vendedor.nome_completo}' inativado com sucesso!") else: messages.warning(request, f"⚠️ Vendedor '{vendedor.nome_completo}' já está inativo.") return redirect('cadastros:vendedor_list') @login_required def vendedor_reativar(request, pk): """ Reativa vendedor inativo POST /cadastros/vendedores//reativar/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') vendedor = get_object_or_404(Vendedor, pk=pk, empresa=empresa) if vendedor.status != 'ATIVO': vendedor.status = 'ATIVO' vendedor.data_desligamento = None vendedor.save() messages.success(request, f"✅ Vendedor '{vendedor.nome_completo}' reativado com sucesso!") else: messages.warning(request, f"⚠️ Vendedor '{vendedor.nome_completo}' já está ativo.") return redirect('cadastros:vendedor_list') # ============================================================================= # API JSON # ============================================================================= @login_required def vendedor_json(request): """ API JSON para listagem de vendedores Útil para autocomplete, select2, AJAX, etc GET /cadastros/vendedores/api/json/ Parâmetros GET opcionais: - status: ATIVO, INATIVO, FERIAS, AFASTADO - tipo_mercado: INTERNO, EXTERNO, AMBOS Retorna: { "vendedores": [ { "id": 1, "nome_completo": "João Silva", "email_corporativo": "joao@empresa.com", "tipo_mercado": "INTERNO", "telefone_principal": "(84) 99999-9999" } ], "total": 1 } """ empresa = request.user.empresa if not empresa: return JsonResponse({'error': 'Empresa não encontrada'}, status=400) # Filtros opcionais via GET status = request.GET.get('status', 'ATIVO') tipo_mercado = request.GET.get('tipo_mercado', '') # QuerySet base vendedores = Vendedor.objects.filter(empresa=empresa) # Aplica filtros if status: vendedores = vendedores.filter(status=status) if tipo_mercado: vendedores = vendedores.filter(tipo_mercado=tipo_mercado) # Retorna apenas campos necessários vendedores_data = list(vendedores.values( 'id', 'nome_completo', 'email_corporativo', 'tipo_mercado', 'telefone_principal' )) return JsonResponse({ 'vendedores': vendedores_data, 'total': len(vendedores_data) }) # ============================================================================= # COMISSÕES # ============================================================================= @login_required def comissao_list(request): """ Lista comissões de vendedores GET /cadastros/comissoes/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') comissoes = ComissaoVendedor.objects.filter( empresa=empresa ).select_related('vendedor').order_by('-referencia_mes') # Filtros opcionais vendedor_id = request.GET.get('vendedor') status = request.GET.get('status') if vendedor_id: comissoes = comissoes.filter(vendedor_id=vendedor_id) if status: comissoes = comissoes.filter(status=status) # Estatísticas total_pendente = comissoes.filter(status='PENDENTE').aggregate( total=Sum('valor_comissao') )['total'] or 0 total_pago = comissoes.filter(status='PAGA').aggregate( total=Sum('valor_comissao') )['total'] or 0 context = { 'comissoes': comissoes, 'total_pendente': total_pendente, 'total_pago': total_pago, 'vendedores': Vendedor.objects.filter(empresa=empresa, status='ATIVO'), } return render(request, 'cadastros/comissao_list.html', context) @login_required def comissao_create(request): """ Cria nova comissão para vendedor GET/POST /cadastros/comissoes/nova/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') if request.method == 'POST': form = ComissaoVendedorForm(request.POST, empresa=empresa) if form.is_valid(): comissao = form.save() messages.success( request, f"✅ Comissão de R$ {comissao.valor_comissao:.2f} criada para " f"{comissao.vendedor.nome_completo}!" ) return redirect('cadastros:comissao_list') else: messages.error(request, '❌ Corrija os erros no formulário.') else: form = ComissaoVendedorForm(empresa=empresa) context = { 'form': form, 'title': 'Nova Comissão' } return render(request, 'cadastros/comissao_form.html', context) @login_required def comissao_pagar(request, pk): """ Marca comissão como paga POST /cadastros/comissoes//pagar/ """ empresa = request.user.empresa if not empresa: messages.error(request, '❌ Você não está vinculado a nenhuma empresa.') return redirect('dashboard:index') comissao = get_object_or_404(ComissaoVendedor, pk=pk, empresa=empresa) if comissao.status == 'PENDENTE': comissao.status = 'PAGA' comissao.data_pagamento = timezone.now().date() comissao.save() messages.success( request, f"✅ Comissão de R$ {comissao.valor_comissao:.2f} marcada como paga!" ) else: messages.warning(request, '⚠️ Esta comissão já foi paga.') return redirect('cadastros:comissao_list') #
